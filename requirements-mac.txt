# Runtime dependencies for the macOS build of Dictate Lite.
# Mirrors requirements-lite.txt (the app is the same Python code) with the
# macOS system-integration packages made explicit so keystroke sending and
# global hotkeys work reliably.

numpy
pynput
pyperclip
pyautogui
Pillow
PySide6

# do nothing.

pyobjc-framework-Cocoa; sys_platform == "darwin"
pyobjc-framework-Quartz; sys_platform == "darwin"
